function build13() { const chart = document.getElementById("chart13"); chart.innerHTML = ""; // Draw the original squares from the worksheet. visibleNumbers.forEach(number => { chart.appendChild(makeSquare(number, "display")); }); // Make all empty hundreds-chart positions clickable. for (let number = 1; number <= 100; number++) { if (visibleNumbers.includes(number)) continue; const pos = position(number); const square = document.createElement("button"); square.type = "button"; square.className = "square target"; square.dataset.number = number; square.style.left = `calc(${pos.col} * var(--cell))`; square.style.top = `calc(${pos.row} * var(--cell))`; // Hide the empty position until selected. square.style.border = "1px solid transparent"; square.style.background = "transparent"; square.textContent = ""; square.setAttribute( "aria-label", "Add a square at this position" ); square.onclick = () => { if (square.classList.contains("active")) { square.classList.remove("active"); square.textContent = ""; square.style.border = "1px solid transparent"; square.style.background = "transparent"; return; } const activeCount = chart.querySelectorAll(".target.active").length; if (activeCount >= 3) { return; } square.classList.add("active"); square.textContent = "+"; square.style.border = "2px solid #36834a"; square.style.background = "#dceedd"; }; chart.appendChild(square); } }
